  if(name==='strategy'){await page.getByText('Available output ceiling',{exact:true}).first().waitFor();await capture('strategy-loaded-mobile');}
 }

 await page.setViewportSize({width:1440,height:1000});await page.evaluate(()=>window.scrollTo(0,0));await page.waitForFunction(()=>!document.querySelector('video').paused);checks.push({name:'video-onscreen',playing:true});
