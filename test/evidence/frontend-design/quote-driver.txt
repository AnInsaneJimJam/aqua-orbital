import {createRequire} from 'node:module';import {writeFile} from 'node:fs/promises';
const {chromium}=createRequire(new URL('../apps/web/package.json',import.meta.url))('@playwright/test');const browser=await chromium.launch({headless:true});const page=await browser.newPage({viewport:{width:1440,height:1000}});const result={signatureSubmitted:false};
try{
 await page.goto('http://127.0.0.1:3000/swap');const connect=page.getByRole('button',{name:'Connect wallet',exact:true}).first();await connect.waitFor();await connect.click();await page.getByRole('button',{name:'Manage connected wallet'}).waitFor();await page.getByLabel('You pay',{exact:true}).fill('1');
 for(let attempt=0;attempt<3;attempt++){
  try{await page.getByRole('button',{name:'Review swap',exact:true}).waitFor({timeout:25000});break;}catch(e){if(attempt===2)throw e;const refresh=page.getByRole('button',{name:'Get quote',exact:true});if(await refresh.isEnabled())await refresh.click();}
 }
 await page.screenshot({path:'.cache/asset-verified/swap-quote.png',fullPage:true});await page.getByRole('button',{name:'Review swap',exact:true}).click();await page.getByRole('button',{name:/Submit reviewed swap|Approve /}).waitFor({timeout:30000});await page.screenshot({path:'.cache/asset-verified/swap-review.png',fullPage:true});result.review=await page.locator('main').innerText();result.passed=true;
}catch(e){result.error=e.message;process.exitCode=1;}finally{await writeFile('.cache/asset-verified/quote-recheck.json',JSON.stringify(result,null,2));console.log(JSON.stringify({...result,review:result.review?'Captured':undefined}));await browser.close();}
