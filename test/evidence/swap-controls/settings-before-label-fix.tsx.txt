'use client';
import type {useSwapSettings} from './useSwapSettings';
export function SwapSettings({state:s}:{state:ReturnType<typeof useSwapSettings>}){
 return <details><summary>Swap settings</summary>
  <fieldset disabled={!s.ready}><legend>Slippage tolerance</legend><div className="row">
   {[10,50,100].map(bps=><button key={bps} type="button" className="button secondary compact" aria-pressed={s.bps===String(bps)} onClick={()=>s.setBps(String(bps))}>{bps/100}%</button>)}
  </div><label className="field">Custom slippage (basis points)<input inputMode="numeric" value={s.bps} onChange={e=>s.setBps(e.target.value)} aria-invalid={s.invalid} aria-describedby="slippage-help"/></label>
  <p id="slippage-help" className="hint">1–500 whole basis points. 100 basis points equals 1%.</p>
  {s.invalid&&<p className="error" role="alert">Choose 1–500 integer basis points and a supported deadline.</p>}
  {s.warning&&<p className="notice" role="status">High slippage: you may receive substantially less than the quoted amount.</p>}
  <label className="field">Transaction deadline<select value={s.seconds} onChange={e=>s.setSeconds(Number(e.target.value))}><option value={60}>1 minute</option><option value={180}>3 minutes</option><option value={600}>10 minutes</option></select></label>
  <button type="button" className="button secondary compact" onClick={s.reset}>Reset settings</button></fieldset>
  {s.message&&<p className="notice" role="status">{s.message}</p>}
 </details>;
}
