pragma solidity 0.8.30;
import {Test} from "forge-std/Test.sol";
import {OrbitalMath as M} from "../src/libraries/OrbitalMath.sol";
import {TickGeometry as G} from "../src/libraries/TickGeometry.sol";
import {CurveEvaluation as C} from "../src/libraries/CurveEvaluation.sol";
import {FrontierEndpoint as E} from "../src/libraries/FrontierEndpoint.sol";
import {CurvePrimitiveFixtures as Scalars} from "./fixtures/CurvePrimitiveFixtures.sol";
import {PilotGasFixtures as Pilot} from "./fixtures/PilotGasFixtures.sol";

contract PrefixDiscoveryTest is Test {
    function legacy(Pilot.Case memory c) private pure returns(E.Identity memory root){
        for(uint8 count;count<c.ticks.length;count++){
            E.Result memory r=E.exactInput(c.x,c.ticks,c.decimals,c.input,c.output,c.raw,count,0);
            if(r.root.identified)return r.root;
        }
    }
    function check(Pilot.Case memory c) private pure {
        bytes32 before=keccak256(abi.encode(c));
        E.Identity memory expected=legacy(c);
        E.Identity memory actual=E.discoverFinal(c.x,c.ticks,c.decimals,c.input,c.output,c.raw);
        assertEq(abi.encode(actual),abi.encode(expected));assertEq(actual.bracket.used,0);assertEq(actual.bracket.remaining,0);
        assertEq(keccak256(abi.encode(c)),before);
    }
    function testConcentratedOutwardDiscoveryMatchesLegacy() public pure {check(Pilot.get(1));}
    function testConcentratedReversalDiscoveryMatchesLegacy() public pure {check(Pilot.get(2));}
    function testConcentratedNoCrossingDiscoveryMatchesLegacy() public pure {check(Pilot.get(3));}
    function testTwoThreeEightTokenDiscoveryMatchesLegacy() public pure {
        uint256[3] memory indices=[uint256(0),2,3];
        for(uint256 k;k<indices.length;k++){
            Pilot.Case memory c;(,c.x,c.ticks,,,,)=Scalars.scalar(indices[k]);
            c.decimals=new uint8[](c.x.length);for(uint256 j;j<c.x.length;j++)c.decimals[j]=18;
            c.input=0;c.output=1;c.raw=1e18;check(c);
        }
    }
    function discover(Pilot.Case memory c) external pure returns(E.Identity memory) {
        return E.discoverFinal(c.x,c.ticks,c.decimals,c.input,c.output,c.raw);
    }
    function testUncertifiedStartDoesNotDiscover() public pure {
        Pilot.Case memory c=Pilot.get(2);for(uint256 k;k<c.x.length;k++)c.x[k]=0;
        check(c);assertFalse(E.discoverFinal(c.x,c.ticks,c.decimals,c.input,c.output,c.raw).identified);
    }
    function testWrongPairStillReverts() public {
        Pilot.Case memory c=Pilot.get(1);c.output=c.input;vm.expectRevert(E.InvalidPair.selector);this.discover(c);
    }
    function testWrongDecimalsStillRevert() public {
        Pilot.Case memory c=Pilot.get(1);c.decimals[2]=19;vm.expectRevert(E.InvalidMetadata.selector);this.discover(c);
    }
    function testZeroAndOversizedInputStillRevert() public {
        Pilot.Case memory c=Pilot.get(1);c.raw=0;vm.expectRevert(E.InvalidInput.selector);this.discover(c);
        c.raw=type(uint256).max;vm.expectRevert(E.InvalidInput.selector);this.discover(c);
    }
    function testFuzzSequentialContextsEqualIndependentPreparation(uint8 nSeed,uint8 ticksSeed,uint192 radiusSeed) public pure {
        uint8 n=uint8(bound(nSeed,2,8));uint8 length=uint8(bound(ticksSeed,1,8));
        uint192 radius=uint192(bound(radiusSeed,1<<104,1<<156));M.Tick[] memory ticks=new M.Tick[](length);
        for(uint256 k;k<length;k++){
            uint64 key=k+1==length?type(uint64).max:uint64((uint256(n)-1)*(1<<32)-(length-k)*(1<<20));
            ticks[k]=M.Tick(key,radius+uint192(k),G.coefficients(n,key));
        }
        C.Context memory ctx=C.prepare(n,ticks,0);
        for(uint8 count;count<length;count++){
            assertEq(abi.encode(ctx),abi.encode(C.prepare(n,ticks,count)));
            if(count+1<length)ctx=C.advance(ctx,ticks);
        }
    }
    function testSequentialPilotContextsMatchAllIndependentPrefixes() public pure {
        Pilot.Case memory c=Pilot.get(1);C.Context memory ctx=C.prepare(8,c.ticks,0);
        for(uint8 count;count<c.ticks.length;count++){
            assertEq(abi.encode(ctx),abi.encode(C.prepare(8,c.ticks,count)));
            if(count+1<c.ticks.length)ctx=C.advance(ctx,c.ticks);
        }
    }
    function advancePastAnchor(M.Tick[] memory ticks) external pure {
        C.Context memory ctx=C.prepare(8,ticks,7);C.advance(ctx,ticks);
    }
    function testContextCannotAdvancePastFullAnchor() public {
        Pilot.Case memory c=Pilot.get(1);vm.expectRevert(C.InvalidContext.selector);this.advancePastAnchor(c.ticks);
    }
}
